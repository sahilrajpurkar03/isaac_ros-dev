    sudo apt install jstest-gtk
    jstest-gtk

    pip3 install pygame

    cp ~/Downloads/joystick_teleop.py ${ISAAC_ROS_WS}/

    python3 joystick_teleop.py

## Step 3: Install Visual SLAM

# open a new terminal (the teleoperation keeps running in the first one) and enter the same container
    cd ${ISAAC_ROS_WS}/src/isaac_ros_common
    ./scripts/run_dev.sh -b -d ${ISAAC_ROS_WS}

# install Visual SLAM (cuVSLAM) inside the container
    sudo apt-get update
    sudo apt-get install -y ros-humble-isaac-ros-visual-slam
    ros2 pkg list | grep visual_slam

# expected output
    isaac_ros_visual_slam
    isaac_ros_visual_slam_interfaces

# note: packages installed with apt inside the container are lost when the container is deleted (it is deleted when you close the last terminal in it). If "ros2 pkg list" shows nothing, run the install commands again

# download sim_cuvslam.launch.py (attached in the description) and put it in a launch folder in your workspace on your PC (not in the container)
# the container sees it at /workspaces/isaac_ros-dev/launch/
    mkdir -p ${ISAAC_ROS_WS}/launch
    cp ~/Downloads/sim_cuvslam.launch.py ${ISAAC_ROS_WS}/launch/

# run the launch file (in this container terminal; the simulation can still be stopped, cuVSLAM waits for the camera images)
    ros2 launch /workspaces/isaac_ros-dev/launch/sim_cuvslam.launch.py
